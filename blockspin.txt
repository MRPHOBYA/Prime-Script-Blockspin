-- ===== ESP Functions =====
local function createHighlight(char)
    if char and not char:FindFirstChild("Hermano_Highlight") then
        local hl = Instance.new("Highlight", char)
        hl.Name = "Hermano_Highlight"
        hl.FillTransparency = 0.92
        hl.OutlineTransparency = 0
        hl.OutlineColor = highlightColor
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Adornee = char
    end
end

local function removeHighlight(char)
    if char then
        local hl = char:FindFirstChild("Hermano_Highlight")
        if hl then hl:Destroy() end
    end
end

local function setupPlayer(plr)
    if plr == Player then return end
    if plr.Character then
        createHighlight(plr.Character)
    end
    plr.CharacterAdded:Connect(function(char)
        if espEnabled then
            createHighlight(char)
        end
    end)
    plr.CharacterRemoving:Connect(removeHighlight)
end

for _, plr in Players:GetPlayers() do
    if espEnabled then
        setupPlayer(plr)
    end
end

Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(function(p)
    removeHighlight(p.Character)
end)

-- ===== Aimbot Helpers =====
local function IsVisible(part)
    if not part or not part.Parent then return false end
    local origin = Camera.CFrame.Position
    local direction = (part.Position - origin)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {Player.Character or {}, part.Parent}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local result = workspace:Raycast(origin, direction, params)
    return result == nil
end

local function GetClosestPlayerToFOV()
    local closest, minDist = nil, FOVRadius
    local center = Camera.ViewportSize / 2
    for _, plr in Players:GetPlayers() do
        if plr ~= Player and plr.Character and plr.Character:FindFirstChild(AimPart) then
            local part = plr.Character[AimPart]
            local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                if dist < minDist and IsVisible(part) then
                    minDist = dist
                    closest = plr
                end
            end
        end
    end
    return closest
end

local function onInputBegan(input, gameProcessed)
    if gameProcessed then return end
    if SuperJumpEnabled and input.KeyCode == Enum.KeyCode.Space then
        local humanoid = Player.Character and Player.Character:FindFirstChildWhichIsA("Humanoid")
        if humanoid then
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
            task.wait()
            humanoid:ChangeState(Enum.HumanoidStateType.Seated)
        end
    end
end

        rightMouseDown = true

    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        rightMouseDown = false
        UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement and rightMouseDown and FreecamEnabled then
        mouseDelta = input.Delta
    end
    if input.UserInputType == Enum.UserInputType.MouseWheel and FreecamEnabled then
        local delta = input.Position.Z
        local newFov = math.clamp(Camera.FieldOfView - delta * ZoomSpeed, MinFOV, MaxFOV)
        Camera.FieldOfView = newFov
    end
end)

-- ===== Main Render Loop =====
local lastTime = tick()
local fps = 0

RunService.RenderStepped:Connect(function(dt)
    -- Stats update
    if fpsPingVisible then
        fps += 1
        if tick() - lastTime >= 1 then
            FPSLabel.Text = "⚡ FPS: " .. fps
            if fps >= 100 then FPSLabel.TextColor3 = Color3.fromRGB(0,255,100)
            elseif fps >= 40 then FPSLabel.TextColor3 = Color3.fromRGB(255,200,60)
            else FPSLabel.TextColor3 = Color3.fromRGB(255,60,60) end
            fps = 0
            lastTime = tick()
        end
        TimeLabel.Text = "🕒 " .. os.date("%H:%M:%S")
        local ping = math.floor(Player:GetNetworkPing() * 1000 + 0.5)
        PingLabel.Text = "🌐 " .. ping .. "ms"
        if ping <= 80 then PingLabel.TextColor3 = Color3.fromRGB(0,255,100)
        elseif ping <= 180 then PingLabel.TextColor3 = Color3.fromRGB(255,215,0)
        else PingLabel.TextColor3 = Color3.fromRGB(220,60,60) end
    end

    FOVCircle.Position = Camera.ViewportSize / 2
    FOVCircle.Radius = FOVRadius
    FOVCircle.Visible = AimbotOn and not FreecamEnabled and not ReconEnabled

    -- Aimbot
    if AimbotOn and not FreecamEnabled and not ReconEnabled then
        local tgt = GetClosestPlayerToFOV()
        if tgt and tgt.Character and tgt.Character:FindFirstChild(AimPart) then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, tgt.Character[AimPart].Position)
        end
    end

    -- Freecam movement
    if FreecamEnabled then
        Camera.CameraType = Enum.CameraType.Scriptable
        if rightMouseDown and mouseDelta.Magnitude > 0 then
            local yaw = -mouseDelta.X * MouseSensitivity
            local pitch = -mouseDelta.Y * MouseSensitivity
            local cf = Camera.CFrame
            local newCf = cf * CFrame.Angles(0, math.rad(yaw), 0) * CFrame.Angles(math.rad(pitch), 0, 0)
            local _, py, pz = newCf:ToEulerAnglesYXZ()
            if math.abs(pz) > math.rad(89) then
                newCf = cf * CFrame.Angles(0, math.rad(yaw), 0)
            end
            Camera.CFrame = newCf
            mouseDelta = Vector2.new()
        end

        local moveDir = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveDir += Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveDir -= Camera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveDir -= Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveDir += Camera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveDir += Vector3.yAxis end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then moveDir -= Vector3.yAxis end
        
        if moveDir.Magnitude > 0.01 then
            local spd = FreecamSpeed
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then spd *= 3 end
            local newPos = Camera.CFrame.Position + moveDir.Unit * spd * dt
            Camera.CFrame = CFrame.new(newPos, newPos + Camera.CFrame.LookVector)
        end
    end

    -- Recon camera follow
    if ReconEnabled and ReconTarget and ReconTarget.Character and ReconTarget.Character:FindFirstChild("Humanoid") then
        local hum = ReconTarget.Character.Humanoid
        if hum.Health > 0 then
            Camera.CameraType = Enum.CameraType.Follow
            Camera.CameraSubject = hum
        else
            ReconTarget = nil
            disableRecon()
        end
    end
end)

local function createInfoLabel(plr)
    if plr == Player then return end
   
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0, 220, 0, 42)
    label.BackgroundTransparency = 1
    label.RichText = true
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
    label.TextColor3 = Color3.fromRGB(255,255,255)
    label.Parent = infoFolder
    label.Visible = false
   
    local connection
    connection = RunService.RenderStepped:Connect(function()
        if not espEnabled or not plr.Parent then
            label:Destroy()
            connection:Disconnect()
            return
        end
       
        local char = plr.Character
        if not char or not char:FindFirstChild("Humanoid") or not char:FindFirstChild("HumanoidRootPart") then
            label.Visible = false
            return
        end
       
        local hum = char.Humanoid
        local root = char.HumanoidRootPart
        local hp = math.floor(hum.Health + 0.5)
        local max = math.floor(hum.MaxHealth)
        local headPos = root.Position + Vector3.new(0, 3.2, 0)
       
        local screenPos, onScreen = Camera:WorldToViewportPoint(headPos)
        local myRoot = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
       
        if not myRoot then
            label.Visible = false
            return
        end
       
        local dist = (root.Position - myRoot.Position).Magnitude
       
        if onScreen and dist < 5000 and hp > 0 then
            label.Position = UDim2.new(0, screenPos.X - 110, 0, screenPos.Y - 60)
            label.Text = string.format(
                "<font color='#ffffff'>%s</font>\n<font color='#ff5555'>%d/%d</font> <font color='#ffff88'>%.0fm</font>",
                plr.Name, hp, max, dist
            )
            label.Visible = true
        else
            label.Visible = false
        end
    end)
end

for _, p in Players:GetPlayers() do
    createInfoLabel(p)
end
